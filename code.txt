1. Write a program to calculate the distance between two points.

#include <iostream>
#include <cmath>
using namespace std;

int main() {
    float x1, y1, x2, y2, distance;
    cout << "Enter first point (x1, y1): ";
    cin >> x1 >> y1;
    cout << "Enter second point (x2, y2): ";
    cin >> x2 >> y2;
    distance = sqrt(pow(x2 - x1, 2) + pow(y2 - y1, 2));
    cout << "Distance between two points = " << distance;

    return 0;
}

3. Write a program to swap two numbers using a temporary variable.

#include <iostream>
using namespace std;

int main() {
    int a, b, temp;

    cout << "Enter two numbers: ";
    cin >> a >> b;

    cout << "\nBefore swapping:";
    cout << "\na = " << a;
    cout << "\nb = " << b;

    // Swapping using temporary variable
    temp = a;
    a = b;
    b = temp;

    cout << "\n\nAfter swapping:";
    cout << "\na = " << a;
    cout << "\nb = " << b;

    return 0;
}

9. Write a program to classify a given number as prime or composite.

#include <iostream>
using namespace std;

int main() {
    int num, i, count = 0;

    cout << "Enter a number: ";
    cin >> num;

    if (num <= 1) {
        cout << num << " is neither Prime nor Composite.";
    }
    else {
        for (i = 1; i <= num; i++) {
            if (num % i == 0) {
                count++;
            }
        }

        if (count == 2) {
            cout << num << " is a Prime Number.";
        }
        else {
            cout << num << " is a Composite Number.";
        }
    }

    return 0;
}

14. Write a program to convert a decimal integer into its binary equivalent.

#include <iostream>
using namespace std;

int main() {
    int num, binary[32], i = 0;

    cout << "Enter a decimal number: ";
    cin >> num;

    if (num == 0) {
        cout << "Binary equivalent = 0";
    }
    else if (num < 0) {
        cout << "Please enter a non-negative integer.";
    }
    else {
        while (num > 0) {
            binary[i] = num % 2;
            num = num / 2;
            i++;
        }

        cout << "Binary equivalent = ";

        for (int j = i - 1; j >= 0; j--) {
            cout << binary[j];
        }
    }

    return 0;
}

16. Write a program to generate the calendar of a month given its starting day and the number of days in that month.

#include <iostream>
#include <iomanip>
using namespace std;

int main() {
    int startDay, days;

    cout << "Enter starting day (0=Sunday, 1=Monday, ..., 6=Saturday): ";
    cin >> startDay;

    cout << "Enter number of days in the month: ";
    cin >> days;

    if (startDay < 0 || startDay > 6 || days < 1 || days > 31) {
        cout << "Invalid input!";
        return 0;
    }

    cout << "\nSun Mon Tue Wed Thu Fri Sat\n";

    // Print spaces before the first date
    for (int i = 0; i < startDay; i++) {
        cout << "    ";
    }

    // Print dates of the month
    for (int date = 1; date <= days; date++) {
        cout << setw(3) << date << " ";

        if ((startDay + date) % 7 == 0) {
            cout << endl;
        }
    }

    cout << endl;

    return 0;
}

20. Write a program to calculate the GCD of two numbers using a recursive function.
#include <iostream>
using namespace std;

// Recursive function to calculate GCD
int findGCD(int a, int b) {
    if (b == 0) {
        return a;
    }

    return findGCD(b, a % b);
}

int main() {
    int num1, num2;

    cout << "Enter two numbers: ";
    cin >> num1 >> num2;

    int gcd = findGCD(num1, num2);

    cout << "GCD of " << num1 << " and "
         << num2 << " = " << gcd << endl;

    return 0;
}
